import React, { useEffect, useState } from "react";
import { Link, useSearchParams } from "react-router-dom";
import { getCourses } from "../lib/courseStore";
import { useLanguage } from "../context/LanguageContext";
import Navbar from "../components/Navbar";
import Footer from "../components/Footer";
import { ArrowUpRight, Filter } from "lucide-react";

export default function Courses() {
  const { language, t } = useLanguage();
  const [courses, setCourses] = useState([]);
  const [loading, setLoading] = useState(true);
  const [activeIndex, setActiveIndex] = useState(0);
  const [params] = useSearchParams();

  const categoriesList = language === "pt"
    ? ["Todos", "Nomadismo", "Idiomas", "Carreira", "Finanças"]
    : ["All", "Nomadism", "Languages", "Career", "Finance"];

  useEffect(() => {
    const ref = params.get("ref");
    if (ref) {
      persistRef(ref);
      api.post("/track/click", { ref_code: ref, source: "courses" }).catch(() => {});
    }
  }, [params]);

  useEffect(() => {
    setLoading(true);
    const rawCategories = ["Todos", "IMERSÃO", "IDIOMAS", "CARREIRA", "FINANÇAS", "COMUNIDADE", "MENTORIA"];
    
    // Fallback translation mapping for UI buttons vs internal categories
    const categoryMapping = {
      "Todos": "Todos", "All": "Todos",
      "Nomadismo": "IMERSÃO", "Nomadism": "IMERSÃO",
      "Idiomas": "IDIOMAS", "Languages": "IDIOMAS",
      "Carreira": "CARREIRA", "Career": "CARREIRA",
      "Finanças": "FINANÇAS", "Finance": "FINANÇAS"
    };

    const uiCategory = categoriesList[activeIndex];
    const backendCat = categoryMapping[uiCategory] || "Todos";

    getCourses().then(allCourses => {
      // Filter by Language
      let filtered = allCourses.filter(c => (c.courseLanguage || 'en') === language);

      // Filter by Category
      if (backendCat !== "Todos") {
        filtered = filtered.filter(c => c.category === backendCat);
      }

      setCourses(filtered);
      setLoading(false);
    });
  }, [activeIndex, language, categoriesList]);

  return (
    <div>
      <Navbar />
      <section data-testid="courses-hero" className="pt-32 pb-10 nomad-container">
        <div className="overline">{t("courses.catalog")}</div>
        <h1 className="font-heading font-black text-5xl md:text-7xl tracking-tighter mt-3">
          {t("courses.title")}
        </h1>
        <p className="text-nomad-muted mt-4 max-w-xl">
          {t("courses.desc")}
        </p>
      </section>

      <section className="nomad-container pb-10">
        <div className="flex flex-wrap items-center gap-2 border-b border-[#E6D7C3] pb-6">
          <Filter size={16} className="text-nomad-muted mr-2" />
          {categoriesList.map((cat, idx) => (
            <button
              key={cat}
              data-testid={`filter-${cat.toLowerCase()}`}
              onClick={() => setActiveIndex(idx)}
              className={`px-4 py-2 rounded-full text-sm font-medium transition ${
                activeIndex === idx
                  ? "bg-[#C05746] text-white"
                  : "bg-white border border-[#E6D7C3] text-nomad-muted hover:text-nomad-ink"
              }`}
            >
              {cat}
            </button>
          ))}
        </div>
      </section>

      <section className="nomad-container pb-24">
        {loading ? (
          <div className="grid sm:grid-cols-2 lg:grid-cols-3 gap-6">
            {Array.from({ length: 6 }).map((_, i) => (
              <div key={i} className="animate-pulse nomad-card">
                <div className="aspect-[4/3] bg-nomad-sand" />
                <div className="p-5 space-y-3">
                  <div className="h-4 bg-nomad-sand rounded w-3/4" />
                  <div className="h-4 bg-nomad-sand rounded w-1/2" />
                </div>
              </div>
            ))}
          </div>
        ) : courses.length === 0 ? (
          <div data-testid="courses-empty" className="text-center py-20 text-nomad-muted">
            {t("courses.empty")}
          </div>
        ) : (
          <div className="grid sm:grid-cols-2 lg:grid-cols-3 gap-6" data-testid="courses-grid">
            {courses.map((c) => (
              <Link
                key={c.id || c.course_id}
                to={`/courses/${c.id || c.course_id}`}
                className="group nomad-card hover:-translate-y-1 transition-all duration-500 hover:shadow-[0_20px_60px_-15px_rgba(192,87,70,0.25)]"
              >
                <div className="aspect-[4/3] relative overflow-hidden bg-[#232F2A]">
                  <img src={c.thumbnailHorizontal || c.thumbnail} alt={c.title} className="w-full h-full object-cover group-hover:scale-105 transition-transform duration-500 opacity-90" />
                  <div className="absolute top-3 left-3 px-3 py-1 rounded-full bg-white/90 backdrop-blur text-xs font-semibold">
                    {c.category}
                  </div>
                </div>
                <div className="p-6">
                  <div className="text-xs uppercase tracking-wider text-nomad-muted mb-2">
                    {c.level || "BEGINNER"} · {c.duration_hours || "10"}{t("courses.hours")}
                  </div>
                  <h3 className="font-heading font-bold text-xl leading-tight group-hover:text-[#C05746]">{c.title}</h3>
                  <p className="text-sm text-nomad-muted mt-2 line-clamp-2">{c.description}</p>
                  <div className="flex items-center justify-between mt-5 pt-5 border-t border-[#E6D7C3]">
                    <div>
                      <div className="font-heading font-black text-2xl mt-1">{c.price || c.formattedPrice}</div>
                    </div>
                    <ArrowUpRight className="text-[#C05746] group-hover:rotate-45 transition-transform" size={22} />
                  </div>
                </div>
              </Link>
            ))}
          </div>
        )}
      </section>

      <Footer />
    </div>
  );
}
